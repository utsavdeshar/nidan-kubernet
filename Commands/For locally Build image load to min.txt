For locally Build image load to minikube cluster:
# On the host
docker tag gateway:latest docker.io/library/gateway:latest
docker save docker.io/library/gateway:latest -o /tmp/gateway.tar


for node in $(kubectl get nodes -o name | cut -d/ -f2); do
  echo "==> $node"
  minikube -p nidan-cluster cp /tmp/gateway.tar "$node":/tmp/gateway.tar
  minikube -p nidan-cluster ssh -n "$node" -- \
    sudo ctr -n k8s.io images import /tmp/gateway.tar
done

Verify:
 for node in $(kubectl get nodes -o name | cut -d/ -f2); do
  printf '%-25s ' "$node"
  minikube -p nidan-cluster ssh -n "$node" -- \
    sudo ctr -n k8s.io images ls | grep -q 'docker.io/library/gateway:latest' \
    && echo OK || echo MISSING
done
