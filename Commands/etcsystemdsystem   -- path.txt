cat /usr/local/bin/nidan-gateway-forward.sh
#!/bin/bash
set -e

PROFILE="nidan-cluster"
LAN_IP="192.168.1.100"
MINIKUBE_IP=$(sudo -u kubernet minikube -p "$PROFILE" ip)

echo "Minikube IP: $MINIKUBE_IP"

socat "TCP-LISTEN:443,bind=${LAN_IP},reuseaddr,fork"  "TCP:${MINIKUBE_IP}:30443" &
socat "TCP-LISTEN:80,bind=${LAN_IP},reuseaddr,fork"   "TCP:${MINIKUBE_IP}:30080" &
socat "TCP-LISTEN:8069,bind=${LAN_IP},reuseaddr,fork" "TCP:${MINIKUBE_IP}:30069" &

wait

sudo chmod +x /usr/local/bin/nidan-gateway-forward.sh   -- privileges




sudo systemctl daemon-reload
sudo systemctl restart nidan-gateway-forward.service
sudo systemctl status  nidan-gateway-forward.service
