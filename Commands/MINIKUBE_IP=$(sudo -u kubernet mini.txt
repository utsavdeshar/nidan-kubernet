To copy .conf file from localhost to kuber pod:
cd ~/nidan
cp nginx.conf nginx.conf.bak        # safety
# Save the file above as nginx.conf

for POD in $(kubectl -n nidan get pod -l app=gateway -o jsonpath='{.items[*].metadata.name}'); do
  echo "==> $POD"
  kubectl -n nidan cp nginx.conf "$POD":/etc/nginx/nginx.conf
  kubectl -n nidan exec "$POD" -- nginx -t
  kubectl -n nidan exec "$POD" -- nginx -s reload
done
