namespace: provide  a layer of isolation in cluster so that it can separate the objects and resources within a cluster using a using different method. By default it create on default namespace if not specified
kubectl get namespaces/ns
kubectl get all --namespace=kube-system or( kubectl get all -n kube-system)   ----list the component with the namespace
kubectl get all --namespace=default

## To create namespace
apiVersion: v1
kind: Namespace
metadata:
  name: demo
kubectl apply -f ns.yaml  ---to create namespace
kubectl get namespace/ns   --list created namespace
kubectl delete ns "namespace"
kubectl create deploy nginx-demo --image=nginx -n demo     ---create a deployment with image nginx within in namespace demo
kubectl get deploy -n demo        ---list the pod deployed within namespace demo

kubectl create deploy nginx-test --image=nginx    create a pod without specifying namespace. It will create the pod in default namespace if not mention -n "namespace" and the same pod name can be created within different namespace
kubectl get deploy  --list the deployed pod in default namespace

kubectl get pods -n=demo  --list the pods in namespace demo
